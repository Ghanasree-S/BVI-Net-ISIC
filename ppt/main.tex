\section{Introduction}
% ============================================================

\begin{frame}{Introduction, Motivation \& Problem Statement}
    \begin{block}{Medical Image Segmentation}
    \begin{itemize}
        \item Helps clinicians localise lesions in dermoscopy, CT and MRI images.
        \item Accuracy pursued by scaling models up: U-Net needs $13.4$M
              parameters, VM-UNet $44.3$M --- too heavy for real clinical hardware.
    \end{itemize}
    \end{block}

    \begin{block}{Problem}
    \begin{itemize}
        \item Blurred lesion boundaries and varying lesion sizes make
              segmentation hard.
        \item Large models overfit on limited medical data.
    \end{itemize}
    \end{block}

    \begin{block}{Task}
    Build an \textbf{ultra-lightweight} segmentation network, accurate on
    \textbf{skin lesion (ISIC2018)} images.
    \end{block}
\end{frame}


\begin{frame}{Base Paper --- BVI-Net}
    BVI-Net imitates the biological visual system: a fast \textbf{dorsal}
    stream for global context and a detail-sensitive \textbf{ventral} stream
    for local features, with the global pathway guiding the local one
    towards the lesion.

    \medskip

    \begin{itemize}
        \item \textbf{Local Pathway:} Gabor-initialised, orientation-selective
              convolution kernels, weights shared across four orientations.
        \item \textbf{Global Pathway (FA-VSSM):} pooling attention gating a
              state space model, scanned four ways for global context.
        \item \textbf{GCN Attention skips:} features projected into a graph
              to link non-adjacent regions during upsampling.
    \end{itemize}

    \medskip
    \alert{Reported: $0.026$M parameters $\mid$ $0.025$ GFLOPs $\mid$ $292$ FPS}

    \bigskip

    \begin{block}{What We Did}
    Reimplemented BVI-Net from scratch in PyTorch (no official code released),
    trained and evaluated on ISIC2018, and compared directly against the
    paper's reported numbers.
    \end{block}
\end{frame}


% ============================================================
\section{Research Gap}
% ============================================================

\begin{frame}{Research Gap}
    Lightweight models already exist, but each trades away something:

    \medskip

    \begin{itemize}
        \item MALUNet ($0.178$M), EGE-UNet ($0.053$M), UltraLight VM-UNet
              ($0.049$M) cut parameters sharply.
        \item CNN-based lightweight models fuse global context poorly.
        \item Mamba/SSM-based models miss local detail on small lesions.
    \end{itemize}

    \medskip

    \begin{block}{Gap Identified}
    No existing model balances global context and local detail at an
    ultra-lightweight scale. BVI-Net claims to, at $0.026$M parameters, but
    \emph{publishes no code} --- leaving the claim unverified.
    \end{block}
\end{frame}

\begin{frame}{Dataset and Preprocessing}
    \begin{block}{Dataset}
    \textbf{ISIC2018 Task 1} --- dermoscopy images with expert-annotated
    lesion masks. Validation uses ISIC's official 100-image set.
    \end{block}

    \medskip

    \begin{block}{Preprocessing}
    \begin{itemize}

\section{Results and Discussion}
% ============================================================

\begin{frame}{Parameter Reduction Journey}
    \begin{table}
        \centering
        \small
        \begin{tabular}{l|c|c}
            \hline
            \textbf{Stage} & \textbf{Params} & \textbf{Val Dice} \\ \hline
            Initial reconstruction & 0.660M & 0.8514 \\ \hline
            + Width search          & 0.037M & 0.8622 \\ \hline
            + True Gabor weight sharing & \textbf{0.0273M} & \textbf{0.8631} \\ \hline
            \textbf{Base paper (claimed)} & \textbf{0.026M} & --- \\ \hline
        \end{tabular}
    \end{table}

    \begin{itemize}
        \item Refactored Gabor convolution into a genuinely shared depthwise
              kernel + pointwise projection --- matching the paper's stated
              mechanism, not just its initialisation.
        \item Result: parameter count within \textbf{5\% of the paper's claim},
              with \textbf{no loss in Dice}.
    \end{itemize}
\end{frame}

\begin{frame}{Results: Ours vs. Base Paper}
    \begin{table}
        \centering
        \small
        \begin{tabular}{l|c|c|c}
            \hline
            \textbf{Metric} & \textbf{Base Paper} & \textbf{Ours} & \textbf{Gap} \\ \hline
            Dice        & 90.43  & 86.31 & $-4.12$ \\ \hline
            mIoU        & 82.45  & 78.18 & $-4.27$ \\ \hline
            Accuracy    & 95.86  & 95.20 & $-0.66$ \\ \hline
            Specificity & 97.15  & 97.09 & $-0.06$ \\ \hline
            Sensitivity & 91.14  & 90.16 & $-0.98$ \\ \hline
            \textbf{Params (M)} & \textbf{0.026} & \textbf{0.0273} & $+5\%$ \\ \hline
        \end{tabular}
        \caption{Skin lesion segmentation, ISIC2018 (all values in \%)}
    \end{table}

    \begin{itemize}
        \item Parameter count matches the paper's claim within 5\%.
        \item Specificity, Accuracy and Sensitivity are within 1 point of the paper.
        \item Dice / mIoU trail by about 4 points (our test set is self-carved:
              ISIC's official test masks are not public).
    \end{itemize}
\end{frame}

\begin{frame}{Liver (LiTS17) and Brain (BraTS19) Results}
    \begin{table}
        \centering
        \small
        \begin{tabular}{l|l|c|c|c}
            \hline
            \textbf{Task} & \textbf{Class} & \textbf{Global Dice} & \textbf{Per-image Dice} & \textbf{Params} \\ \hline
            Liver & Liver & \textbf{92.84} & 82.75 & \multirow{2}{*}{0.0273M} \\
                  & Tumor & 56.65 & 66.14 & \\ \hline
            Brain & WT    & \textbf{86.15} & 70.53 & \multirow{3}{*}{0.0273M} \\
            (LGG) & TC    & 71.06 & 59.33 & \\
                  & ET    & 74.44 & 75.33 & \\ \hline
        \end{tabular}
        \caption{Held-out test sets, patient-level 7:1:2 splits (values in \%)}
    \end{table}
    \begin{itemize}
        \item \textbf{Global Dice} pools all test pixels per class --- trustworthy for
              small classes, since per-image Dice scores a tumor-free slice as 1.0
              when nothing is predicted.
        \item Same 27K-parameter network for all three organs (1, 3 or 4 input channels).
    \end{itemize}
\end{frame}

\begin{frame}{Fixing Small-Class Collapse}
    \begin{itemize}
        \item First liver run: tumor \textbf{sensitivity 0.00} --- the model never
              predicted tumor; brain ET sensitivity 0.10.
        \item Cause: \textbf{per-sample Dice loss}. On every tumor-free slice only an
              exactly empty prediction scores well, so ``predict nothing'' wins.
              Per-batch validation Dice then scored those batches as 1.0 and
              stopped training early (epoch 12).
        \item Fix: Dice pooled \textbf{per class over the batch}; validation and early
              stopping on \textbf{dataset-level} Dice; more liver slices (stride 5).
    \end{itemize}
    \begin{table}
        \centering
        \small
        \begin{tabular}{l|c|c}
            \hline
            \textbf{Test metric} & \textbf{Before} & \textbf{After} \\ \hline
            Liver global Dice         & ---   & \textbf{92.84} \\ \hline
            Tumor Dice (per-image)    & 61.88$^\ast$ & \textbf{66.14} \\ \hline
            Tumor sensitivity         & 0.00  & \textbf{16.19} \\ \hline
            Brain ET Dice (per-image) & 57.21 & \textbf{75.33} \\ \hline
        \end{tabular}
    \end{table}
    {\scriptsize $^\ast$ per-image Dice of a model that never predicts tumor --- inflated by tumor-free slices scored as 1.0.}
\end{frame}

\begin{frame}{Ablation Study (ISIC2018)}
    \begin{table}
        \centering
        \small
        \begin{tabular}{l|c|c|c}
            \hline
            \textbf{Variant} & \textbf{Params} & \textbf{Test Dice} & $\Delta$ \\ \hline
            \textbf{Full BVI-Net} & 27,312 & \textbf{86.31} & --- \\ \hline
            w/o GCN-attention skips & 25,360 & 84.56 & $-1.75$ \\ \hline
            w/o FA-VSSM global pathway & 16,305 & 85.70 & $-0.61$ \\ \hline
            w/o Gabor bank (plain conv) & 29,272 & 86.98 & $+0.67$ \\ \hline
        \end{tabular}
    \end{table}
    \begin{itemize}
        \item \textbf{GCN skips} matter most, then the global pathway.
        \item A plain convolution matched the Gabor bank (+0.67, within run-to-run
              noise) with 7\% more parameters --- the Gabor bank buys compactness.
    \end{itemize}
\end{frame}

\begin{frame}{Deployment: Web Dashboard}
    \begin{itemize}
        \item \textbf{FastAPI backend} serves all three checkpoints; \textbf{React}
              dashboard: pick organ, upload scan, view mask + overlay, test-set scores,
              per-image confidence, latency, parameters and GFLOPs.
        \item Kaggle checkpoints use CUDA \texttt{mamba\_ssm}; for CPU inference we wrote a
              \textbf{parameter-compatible pure-PyTorch Mamba} (vectorised log-space
              selective scan).
        \item CPU masks match the Kaggle GPU predictions \textbf{pixel-for-pixel}.
        \item Speed (batch 1, Tesla T4): \textbf{87 / 84 / 44 FPS} (skin / brain / liver) after
              batching the 4 scan directions into one Mamba call (1.5--1.8\$\times\$ faster);
              paper: 292 FPS.

\section{Conclusion}
% ============================================================

\begin{frame}{Conclusion}
    \begin{itemize}
        \item BVI-Net reimplemented with the real Mamba selective scan and true
              Gabor weight sharing, trained on \textbf{all three tasks}: skin
              \textbf{86.31} Dice, liver \textbf{92.84} and brain WT \textbf{86.15}
              global Dice.

        \medskip

        \item Batch-pooled Dice fixed the small-class collapse (liver tumor, brain ET).

        \medskip

        \item Deployed as a working web dashboard with CPU inference.

        \medskip

        \item Parameter count (\textbf{0.0273M}) matches the claimed
              \textbf{0.026M} within 5\% --- the ultra-lightweight claim is
              \textbf{verified}.
    \end{itemize}

    \bigskip

    \begin{block}{Key Takeaway}
    The dual-pathway, bio-inspired design \textbf{is reproducible, effective,
    and genuinely ultra-lightweight} once the paper's weight-sharing
    mechanism is implemented faithfully.
    \end{block}
\end{frame}


\begin{frame}{Future Work}
    \begin{itemize}
        \item Improve liver \textbf{tumor} segmentation (56.65 global Dice; tumor-slice
              oversampling helped only slightly): all labelled slices, liver-cropped tumor stage.
        \item Ablation study: isolate contribution of Gabor weight sharing,
              FA-VSSM, and GCN-Attention skips individually.
        \item Close the remaining speed gap to 292 FPS (fp16, CUDA graphs); BraTS HGG cases;
              multi-seed ablations.
    \end{itemize}
\end{frame}
